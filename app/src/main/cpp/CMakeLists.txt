cmake_minimum_required(VERSION 3.22)
project(G1AwakeningAndroid LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(G1_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../..")
get_filename_component(G1_ROOT "${G1_ROOT}" ABSOLUTE)
set(G1_UPSTREAM "${G1_ROOT}/upstream/ios/native")
if(ANDROID)
  find_library(G1_ZLIB z REQUIRED)
  set(ZLIB_LIBRARY "${G1_ZLIB}" CACHE FILEPATH "Android system zlib" FORCE)
  set(ZLIB_INCLUDE_DIR "${CMAKE_SYSROOT}/usr/include" CACHE PATH "Android zlib headers" FORCE)
endif()
set(BUILD_TESTING OFF CACHE BOOL "Portable tests run separately on host" FORCE)
add_subdirectory("${G1_UPSTREAM}" "${CMAKE_CURRENT_BINARY_DIR}/ios-core" EXCLUDE_FROM_ALL)
add_library(g1awakening SHARED JNI.cpp AndroidGame.cpp AndroidAudio.cpp AndroidUI.cpp
  AssetStore.cpp GLESRenderer.cpp GLESModelRenderer.cpp)
target_link_libraries(g1awakening PRIVATE g1_resources android log EGL GLESv3 aaudio)
target_compile_options(g1awakening PRIVATE -Wall -Wextra -ffp-contract=off)
target_link_options(g1awakening PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
target_compile_definitions(g1awakening PRIVATE G1_IOS_SOURCE_COMMIT="be203a911bc26313615ea450758413e5c9cf6462")
